	//views::iota(1, 1000)을 하면 1~1000의 숫자가 저장된 오브젝트가 만들어지는게 아니다! 하나씩 데이터를 만들어내서 변수에 넘겨주는 것이다.
	//그래서 views를 사용해서 변수를 생성할 때는 auto로 RValue로 받는 것이 좋다.
	for (auto&& num :
		views::iota(1, 1000)
		| views::filter(odd)
		| views::filter(ninemultiples))
		{
			cout << num << " ";
		}
		
	cout << endl;
}
